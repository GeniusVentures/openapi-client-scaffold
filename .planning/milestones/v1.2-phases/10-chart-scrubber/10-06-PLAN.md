    class _RangeBandPainter extends CustomPainter {
      _RangeBandPainter({
        required this.startPixel,
        required this.endPixel,
        required this.bandColor,       // palette.lightGreenPrimary.withAlpha(0.12)
        required this.edgeColor,       // palette.lightGreenPrimary
        required this.plotHeight,
      });

      @override
      void paint(Canvas canvas, Size size) {
        final double left = min(startPixel.dx, endPixel.dx);
        final double right = max(startPixel.dx, endPixel.dx);
        // Fill
        canvas.drawRect(
          Rect.fromLTRB(left, 0, right, plotHeight),
          Paint()..color = bandColor,
        );
        // Edge rules (1px)
        final Paint edgePaint = Paint()
          ..color = edgeColor
          ..strokeWidth = 1.0;
        canvas.drawLine(Offset(left, 0), Offset(left, plotHeight), edgePaint);
        canvas.drawLine(Offset(right, 0), Offset(right, plotHeight), edgePaint);
      }

      @override
      bool shouldRepaint(_RangeBandPainter old) =>
          startPixel != old.startPixel || endPixel != old.endPixel;
    }
